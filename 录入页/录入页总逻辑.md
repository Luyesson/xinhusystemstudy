function openxiangs(name, num, id, cbal) {
  if (!id) id = 0;
  if (!cbal) cbal = "";
  var url = "task.php?a=p&num=" + num + "&mid=" + id + "";

  var jg = num.indexOf("?") > -1 ? "&" : "?";
  if (num.indexOf("?") > -1 || num.substr(0, 4) == "http") {
    url = num + "" + jg + "callback=" + cbal + "";
  } else {
    url += "&callback=" + cbal + "";
  }

  js.winiframe(name, url);
  return false;
}
